cmake_minimum_required(VERSION 3.16)
project(muon_beam LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Geant4 REQUIRED)
include(${Geant4_USE_FILE})
find_package(ROOT REQUIRED COMPONENTS Core RIO Hist Graf Gpad)
add_executable(muon_beam muon_beam.cc)
target_link_libraries(muon_beam PRIVATE ${Geant4_LIBRARIES} ROOT::Core ROOT::RIO ROOT::Hist ROOT::Graf ROOT::Gpad)
