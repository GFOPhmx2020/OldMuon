cmake_minimum_required(VERSION 3.16)
project(muon_stalactite_full LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
option(SAMPLER_ONLY "Build dependency-free sampler self-test only" OFF)
add_executable(stalactite_sampler_test muon_stalactite_full.cc)
target_compile_definitions(stalactite_sampler_test PRIVATE STALACTITE_SELF_TEST)
enable_testing()
add_test(NAME sampler COMMAND stalactite_sampler_test)
if(NOT SAMPLER_ONLY)
  find_package(Geant4 REQUIRED)
  include(${Geant4_USE_FILE})
  find_package(ROOT REQUIRED COMPONENTS Core RIO Hist Graf Gpad)
  add_executable(muon_stalactite_full muon_stalactite_full.cc)
  target_link_libraries(muon_stalactite_full PRIVATE ${Geant4_LIBRARIES} ROOT::Core ROOT::RIO ROOT::Hist ROOT::Graf ROOT::Gpad)
  configure_file(materials.tsv materials.tsv COPYONLY)
endif()
